CREATE TABLE `compra` (
    `compra_id` INT NOT NULL AUTO_INCREMENT,
    `compra_codigo` VARCHAR(100) NOT NULL,
    `compra_fecha` DATE NOT NULL,
    `compra_hora` TIME NOT NULL,
    `compra_total` DECIMAL(30,2) NOT NULL DEFAULT 0.00,
    `compra_pagado` DECIMAL(30,2) NOT NULL DEFAULT 0.00,
    `compra_cambio` DECIMAL(30,2) NOT NULL DEFAULT 0.00,
    `usuario_id` INT NOT NULL,
    `proveedor_id` INT NULL,
    PRIMARY KEY (`compra_id`),
    UNIQUE KEY `compra_codigo` (`compra_codigo`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
